"""Archive this session's reproducible experiments; never copy raw data or checkpoints."""
from pathlib import Path
import hashlib
import json
import shutil
import subprocess
import time

work = Path(__file__).resolve().parent
repo = work / 'draw-fused'
archive = repo / 'docs' / 'validation' / 'refinement-checkpoint-20260911'
archive.mkdir(parents=True, exist_ok=True)
inventory = []
for name in ['draw-fix', 'draw-fix-fast', 'draw-neural100', 'halfkp-trial',
             'halfkp-anneal-128', 'halfkp-width-64', 'halfkp-refine-128', 'no-ponder-trial']:
    directory = work / name
    destination = archive / name
    destination.mkdir(exist_ok=True)
    head = subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=directory, text=True).strip()
    patch = subprocess.check_output(['git', 'diff', '--binary', 'HEAD', '--', '*.py',
                                     'pyproject.toml'], cwd=directory)
    (destination / 'uncommitted-code.patch').write_bytes(patch)
    for relative in ['chess_draw.py', 'tools/draw_regression.py'] if name == 'draw-fix' else []:
        source = directory / relative
        if source.exists():
            shutil.copyfile(source, destination / (relative.replace('/', '_') + '.txt'))
    assets = []
    for relative in ['weights/halfkp.npz', 'book/codekiddy.bin']:
        source = directory / relative
        if source.exists():
            assets.append({'path': relative, 'bytes': source.stat().st_size,
                           'sha256': hashlib.sha256(source.read_bytes()).hexdigest()})
    inventory.append({'worktree': name, 'head': head, 'archived_patch_bytes': len(patch),
                      'assets_manifest_only': assets})
    if name in ['draw-fix', 'no-ponder-trial']:
        candidates = list(directory.glob('data-*.log'))
        candidates += list((directory / 'data' / 'runs').glob('*.log'))
        candidates += list((directory / 'data' / 'runs').glob('*.json'))
        for source in candidates:
            if source.stat().st_size < 2_000_000:
                shutil.copyfile(source, destination / source.name)

scratch = archive / 'reproduction'
scratch.mkdir(exist_ok=True)
for source in work.glob('*.py'):
    shutil.copyfile(source, scratch / (source.name + '.txt'))
for source in work.glob('*.json'):
    if source.stat().st_size < 2_000_000:
        shutil.copyfile(source, archive / source.name)
for source in (repo / 'data' / 'runs').glob('fused-*'):
    if source.is_file() and source.stat().st_size < 2_000_000:
        shutil.copyfile(source, archive / source.name)
(archive / 'inventory.json').write_text(json.dumps(inventory, indent=2) + '\n')
