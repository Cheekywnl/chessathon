"""Independent fixed-depth prototype of our own search in Numba; not a submission.

The move generator, evaluator and pruning rules come from this team's current
engine. This prototype must match that engine before any integration is considered.
"""
import json
import sys
import time
from pathlib import Path

import numpy as np
from numba import njit, types
from numba.experimental import jitclass
from numba.typed import Dict

WORK = Path(__file__).resolve().parent
sys.path.insert(0, str(WORK / 'draw-wide256'))
import chess
import chess_eval as ce
import chess_halfkp_int as qi
import chess_movegen as mg
import chess_search as ref
import chess_state as cst

MATE, MATE_THRESHOLD, NO_MOVE = ref.MATE, ref.MATE_THRESHOLD, ref.NO_MOVE
CONTEMPT, MAX_PLY = ref.CONTEMPT, ref.MAX_PLY
VALUES, LMR = ref.PIECE_VALUES, ref.LMR_TABLE
U1 = np.uint64(1)

spec = [
    ('keys', types.uint64[::1]), ('depths', types.int16[::1]),
    ('scores', types.int32[::1]), ('flags', types.int8[::1]), ('moves', types.int32[::1]),
    ('mask', types.int64), ('seen', types.DictType(types.uint64, types.int64)),
    ('killers', types.int64[:, ::1]), ('history', types.int64[:, :, ::1]),
    ('nodes', types.int64), ('limit', types.int64), ('params', types.int32[::1]),
    ('w1', types.int16[:, ::1]), ('b1', types.int16[::1]),
    ('w2', types.int8[:, ::1]), ('b2', types.int32[::1]),
    ('w3', types.int8[:, ::1]), ('b3', types.int32[::1]),
    ('w4', types.int8[::1]), ('b4', types.int64), ('scale2', types.int64),
    ('scale3', types.int64), ('divisor', types.float64), ('blend', types.int64),
]


@jitclass(spec)
class Context:
    def __init__(self, power, seen, params, w1, b1, w2, b2, w3, b3, w4, b4, s2, s3, div, blend):
        n = 1 << power
        self.keys = np.zeros(n, dtype=np.uint64)
        self.depths = np.zeros(n, dtype=np.int16)
        self.scores = np.zeros(n, dtype=np.int32)
        self.flags = np.full(n, -1, dtype=np.int8)
        self.moves = np.full(n, -1, dtype=np.int32)
        self.mask = n - 1
        self.seen = seen
        self.killers = np.full((MAX_PLY, 2), -1, dtype=np.int64)
        self.history = np.zeros((2, 64, 64), dtype=np.int64)
        self.nodes, self.limit = 0, 100_000_000
        self.params = params
        self.w1, self.b1, self.w2, self.b2 = w1, b1, w2, b2
        self.w3, self.b3, self.w4, self.b4 = w3, b3, w4, b4
        self.scale2, self.scale3, self.divisor, self.blend = s2, s3, div, blend


@njit
def key_of(s):
    return cst.zobrist_hash(s[0],s[1],s[2],s[3],s[4],s[5],s[6],s[7],s[8],s[9],s[10],
                           cst.ZOBRIST_PIECE_SQUARE,cst.ZOBRIST_CASTLING,
                           cst.ZOBRIST_EP_FILE,cst.ZOBRIST_TURN)


@njit
def legal(s):
    return mg.generate_legal_moves_bb(s[0],s[1],s[2],s[3],s[4],s[5],s[6],s[7],
                                     s[8],s[9],s[10])


@njit
def check(s):
    own = s[6] if s[8] else s[7]
    king = cst._lsb_index(s[5] & own)
    return mg.is_square_attacked(king,not s[8],s[0],s[1],s[2],s[3],s[4],s[5],s[6],s[7])


@njit
def after(s,key,f,t,p):
    return cst.make_move_info(s,key,f,t,p)


@njit
def capture(s,f,t):
    return bool(((s[6]|s[7]) & (U1 << np.uint64(t)))
                or ((s[0] & (U1 << np.uint64(f))) and t == s[10]))


@njit
def piece_at(s,square):
    mask = U1 << np.uint64(square)
    pieces = (s[0],s[1],s[2],s[3],s[4],s[5])
    for i in range(6):
        if pieces[i] & mask:
            return i + 1
    return 6


@njit
def victim_at(s,f,t):
    if t == s[10] and (s[0] & (U1 << np.uint64(f))):
        return 1
    kind = piece_at(s,t)
    return 1 if kind == 6 else kind


@njit
def see(s,f,t):
    return cst.see_raw(s[0],s[1],s[2],s[3],s[4],s[5],s[6],s[7],f,t,s[10],s[8],VALUES)


@njit
def bare(s):
    major = s[3] | s[4]
    nonking = s[0] | s[1] | s[2] | major
    return bool((major & s[6]) and not (nonking & s[7])) or bool(
        (major & s[7]) and not (nonking & s[6]))


@njit
def insufficient_side(own,other,s):
    if own & (s[0] | s[3] | s[4]):
        return False
    if own & s[1]:
        return ce._popcount(own) <= 2 and not (other & ~s[5] & ~s[4])
    if own & s[2]:
        same = not (s[2] & np.uint64(0xAA55AA55AA55AA55)) or not (
            s[2] & np.uint64(0x55AA55AA55AA55AA))
        return same and not s[0] and not s[1]
    return True


@njit
def is_draw(s,key,ctx):
    if s[11] >= 100 or ctx.seen.get(key,0) >= 3:
        return not (check(s) and legal(s)[3] == 0)
    if s[11] == 99:
        f,t,p,n = legal(s)
        for i in range(n):
            child,_,_ = after(s,key,int(f[i]),int(t[i]),int(p[i]))
            if child[11] >= 100 and legal(child)[3] > 0:
                return True
    if ce._popcount(s[6]|s[7]) <= 6:
        return insufficient_side(s[6],s[7],s) and insufficient_side(s[7],s[6],s)
    return False


@njit
def draw_score(ply):
    return -CONTEMPT if ply % 2 == 0 else CONTEMPT


@njit
def evaluate(s,mobility,ctx):
    wc = int(bool(s[9] & U1)) + int(bool(s[9] & np.uint64(128)))
    bc = int(bool(s[9] & np.uint64(1 << 56))) + int(bool(s[9] & np.uint64(1 << 63)))
    positional = ce.evaluate(s[0],s[1],s[2],s[3],s[4],s[5],s[6],s[7],ce.MG_PST,ce.EG_PST,
                            ce.PASSED_MASK_WHITE,ce.PASSED_MASK_BLACK,ce.FILE_MASK,
                            ce.ADJACENT_FILE_MASK,wc,bc,ctx.params)
    classical = (int(positional) if s[8] else -int(positional)) + int(ctx.params[ce.P_MOBILITY])*mobility
    nonking = s[0]|s[1]|s[2]|s[3]|s[4]
    if ctx.blend <= 0 or ce._popcount(s[6]|s[7]) < 8 or not (nonking&s[6]) or not (nonking&s[7]):
        return classical
    neural = qi.evaluate(s[0],s[1],s[2],s[3],s[4],s[5],s[6],s[7],s[8],ctx.w1,ctx.b1,
                         ctx.w2,ctx.b2,ctx.w3,ctx.b3,ctx.w4,ctx.b4,ctx.scale2,ctx.scale3,ctx.divisor)
    bonus = int(ce.mop_up_bonus(s[0],s[1],s[2],s[3],s[4],s[5],s[6],s[7],ctx.params))
    neural = max(-20000,min(20000,neural)) + (bonus if s[8] else -bonus)
    return neural if ctx.blend >= 100 else round((classical*(100-ctx.blend)+neural*ctx.blend)/100)


@njit
def pack(f,t,p):
    return np.int64(f | (t << 6) | (p << 12))


@njit
def unpack(move):
    return move & 63, (move >> 6) & 63, (move >> 12) & 7


@njit
def ordered(s,f,t,p,n,ctx,ttmove,ply,qmode,incheck):
    moves = np.empty(n,dtype=np.int64)
    captures = np.empty(n,dtype=np.bool_)
    scores = np.empty(n,dtype=np.int64)
    used = 0
    kp = min(ply,MAX_PLY-1)
    for i in range(n):
        ff,tt,pp = int(f[i]),int(t[i]),int(p[i])
        move = pack(ff,tt,pp)
        cap = capture(s,ff,tt)
        if qmode:
            if not incheck and not pp and not cap:
                continue
            score = 200000 if pp else VALUES[victim_at(s,ff,tt)-1]*10 - VALUES[piece_at(s,ff)-1]
        elif move == ttmove:
            score = 1000000
        elif cap:
            exchange = see(s,ff,tt)
            score = 100000+exchange if exchange >= 0 else -100000+exchange
        elif move == ctx.killers[kp,0]:
            score = 90000
        elif move == ctx.killers[kp,1]:
            score = 89000
        else:
            score = ctx.history[int(s[8]),ff,tt]
        moves[used],captures[used],scores[used] = move,cap,score
        used += 1
    # Strict '<' preserves the original legal-generation order when scores tie.
    for i in range(1,used):
        move,cap,score = moves[i],captures[i],scores[i]
        j = i - 1
        while j >= 0 and scores[j] < score:
            moves[j+1],captures[j+1],scores[j+1] = moves[j],captures[j],scores[j]
            j -= 1
        moves[j+1],captures[j+1],scores[j+1] = move,cap,score
    return moves,captures,used


@njit
def probe(ctx,key,depth,alpha,beta,ply):
    i = np.int64(key & np.uint64(ctx.mask))
    if ctx.flags[i] < 0 or ctx.keys[i] != key:
        return 0,NO_MOVE,False
    move = np.int64(ctx.moves[i])
    if ctx.depths[i] >= depth:
        score = np.int64(ctx.scores[i])
        if score >= MATE_THRESHOLD:
            score -= ply
        elif score <= -MATE_THRESHOLD:
            score += ply
        flag = ctx.flags[i]
        if flag == 0 or (flag == 1 and score >= beta) or (flag == 2 and score <= alpha):
            return score,move,True
    return 0,move,False


@njit
def put(ctx,key,depth,score,flag,move,ply):
    i = np.int64(key & np.uint64(ctx.mask))
    if ctx.flags[i] < 0 or ctx.keys[i] == key or ctx.depths[i] <= depth:
        if score >= MATE_THRESHOLD:
            score += ply
        elif score <= -MATE_THRESHOLD:
            score -= ply
        ctx.keys[i],ctx.depths[i],ctx.scores[i],ctx.flags[i],ctx.moves[i] = key,depth,score,flag,move


@njit
def tick(ctx):
    ctx.nodes += 1
    if ctx.nodes > ctx.limit:
        raise RuntimeError('prototype node budget exceeded')


@njit
def qsearch(s,alpha,beta,ply,key,incheck,ctx):
    tick(ctx)
    if is_draw(s,key,ctx):
        return draw_score(ply)
    f,t,p,n = legal(s)
    if n == 0:
        return -(MATE-ply) if incheck else draw_score(ply)
    stand = 0
    if incheck:
        best = -MATE-1
    else:
        stand = evaluate(s,n,ctx)
        if stand >= beta:
            return stand
        alpha = max(alpha,stand)
        best = stand
    moves,caps,count = ordered(s,f,t,p,n,ctx,NO_MOVE,ply,True,incheck)
    for i in range(count):
        ff,tt,pp = unpack(moves[i])
        if not incheck and caps[i] and not pp:
            victim = victim_at(s,ff,tt)
            if stand + VALUES[victim-1] + ref.DELTA_MARGIN <= alpha:
                continue
            if see(s,ff,tt) < 0:
                continue
        child,childkey,childcheck = after(s,key,ff,tt,pp)
        ctx.seen[childkey] = ctx.seen.get(childkey,0) + 1
        score = -qsearch(child,-beta,-alpha,ply+1,childkey,childcheck,ctx)
        ctx.seen[childkey] -= 1
        best = max(best,score)
        alpha = max(alpha,score)
        if alpha >= beta:
            break
    return best


@njit
def negamax(s,depth,alpha,beta,ply,allow_null,key,incheck,ctx):
    tick(ctx)
    if is_draw(s,key,ctx):
        return draw_score(ply)
    score,ttmove,hit = probe(ctx,key,depth,alpha,beta,ply)
    if hit:
        return score
    if depth <= 0:
        return qsearch(s,alpha,beta,ply,key,incheck,ctx)
    f,t,p,n = legal(s)
    if n == 0:
        return -(MATE-ply) if incheck else draw_score(ply)
    static = 0
    if not incheck:
        static = evaluate(s,n,ctx)
        if depth <= ref.REVERSE_FUTILITY_DEPTH and abs(beta) < MATE_THRESHOLD:
            margin = ref.REVERSE_FUTILITY_MARGIN_PER_PLY*depth
            if static-margin >= beta:
                return static-margin
    own = s[6] if s[8] else s[7]
    if allow_null and not incheck and depth >= 3 and (own & ~s[0] & ~s[5]):
        null = (s[0],s[1],s[2],s[3],s[4],s[5],s[6],s[7],not s[8],s[9],-1,s[11]+1)
        score = -negamax(null,depth-1-ref.NULL_MOVE_REDUCTION,-beta,-beta+1,ply+1,
                         False,key_of(null),check(null),ctx)
        if score >= beta:
            return score
    moves,caps,count = ordered(s,f,t,p,n,ctx,ttmove,ply,False,incheck)
    original_alpha = alpha
    best,bestmove = -MATE-1,NO_MOVE
    kp = min(ply,MAX_PLY-1)
    prunable = not bare(s)
    for i in range(count):
        move,cap = moves[i],caps[i]
        ff,tt,pp = unpack(move)
        extension = 1 if incheck else 0
        killer = move == ctx.killers[kp,0] or move == ctx.killers[kp,1]
        child,childkey,childcheck = after(s,key,ff,tt,pp)
        if (prunable and depth <= ref.LATE_MOVE_PRUNING_DEPTH
                and i >= ref.LATE_MOVE_PRUNING_BASE+ref.LATE_MOVE_PRUNING_PER_DEPTH*depth
                and not extension and not cap and not pp and not killer
                and best > -MATE_THRESHOLD and not childcheck):
            continue
        if (prunable and not incheck and i >= 1 and depth <= ref.FUTILITY_DEPTH
                and not extension and not cap and not pp and not killer
                and abs(alpha) < MATE_THRESHOLD
                and static+ref.FUTILITY_MARGIN_PER_PLY*depth <= alpha and not childcheck):
            continue
        reduction = 0
        if prunable and depth >= 3 and i >= 3 and not extension and not cap and not pp and not killer:
            reduction = min(int(LMR[min(depth,ref._LMR_MAX_DEPTH),min(i,ref._LMR_MAX_MOVE_INDEX)]),depth-1)
        ctx.seen[childkey] = ctx.seen.get(childkey,0) + 1
        if i == 0:
            score = -negamax(child,depth-1+extension,-beta,-alpha,ply+1,True,childkey,childcheck,ctx)
        else:
            score = -negamax(child,depth-1+extension-reduction,-alpha-1,-alpha,ply+1,
                             True,childkey,childcheck,ctx)
            if score > alpha:
                score = -negamax(child,depth-1+extension,-beta,-alpha,ply+1,True,childkey,childcheck,ctx)
        ctx.seen[childkey] -= 1
        if score > best:
            best,bestmove = score,move
        alpha = max(alpha,score)
        if alpha >= beta:
            if not cap:
                if move != ctx.killers[kp,0]:
                    ctx.killers[kp,1] = ctx.killers[kp,0]
                    ctx.killers[kp,0] = move
                ctx.history[int(s[8]),ff,tt] += depth*depth
            break
    flag = 2 if best <= original_alpha else 1 if best >= beta else 0
    put(ctx,key,depth,best,flag,bestmove,ply)
    return best


def context(history=None):
    weights = ref.HALFKP_WEIGHTS
    seen = Dict.empty(key_type=types.uint64,value_type=types.int64)
    for key,value in (history or {}).items():
        seen[np.uint64(key)] = np.int64(value)
    return Context(16,seen,ce.DEFAULT_PARAMS,weights.w1,weights.b1,weights.w2,weights.b2,
                   weights.w3,weights.b3,weights.w4,weights.b4,weights.scale2,weights.scale3,
                   weights.output_divisor,ref.HALFKP_BLEND)


def main():
    import psutil
    psutil.Process().cpu_affinity([14])
    import agent
    agent._stop_pondering()
    state = ref.state_from_board(chess.Board())
    ctx = context()
    ctx.limit = 2_000_000
    started = time.perf_counter()
    print('Compiling fixed-depth prototype',flush=True)
    value = negamax(state,2,-MATE,MATE,0,True,np.uint64(key_of(state)),check(state),ctx)
    compile_seconds = time.perf_counter()-started
    print(json.dumps({'compile_seconds':compile_seconds,'warm_score':value,'warm_nodes':ctx.nodes}),flush=True)
    data=json.loads((WORK/'halfkp-trial/data/runs/openings-final-20260914.json').read_text())
    boards=[chess.Board(p['fen']) for p in data['positions'][:10]]
    boards += [chess.Board(fen) for fen in (
        'r3k2r/8/8/8/8/8/8/R3K2R w KQkq - 0 1',
        '4k3/8/8/3pP3/8/8/8/4K3 w - d6 0 1',
        '1r2k3/P7/8/8/8/8/7p/4K3 w - - 0 1',
        '1K1k4/1P6/8/8/8/8/r7/2R5 w - - 0 1',
        '7k/5K2/6Q1/8/8/8/8/8 b - - 0 90',
    )]
    results=[]
    for index,board in enumerate(boards):
        state=ref.state_from_board(board)
        key=ref.hash_of(state)
        before=ref.Search(ref.TranspositionTable(16),{key:1})
        before.deadline=time.monotonic()+120
        ctx=context({key:1});ctx.limit=2_000_000
        for implementation in ((0,1) if index % 2 == 0 else (1,0)):
            started=time.perf_counter()
            if implementation == 0:
                expected=before.negamax(state,6,-MATE,MATE,0)
                old_seconds=time.perf_counter()-started
            else:
                actual=negamax(state,6,-MATE,MATE,0,True,np.uint64(key),check(state),ctx)
                new_seconds=time.perf_counter()-started
        record={'index':index,'fen':board.fen(),'before_score':expected,'after_score':actual,
                'before_nodes':before.nodes,'after_nodes':ctx.nodes,
                'before_seconds':old_seconds,'after_seconds':new_seconds}
        print(json.dumps(record),flush=True)
        assert actual == expected and ctx.nodes == before.nodes,record
        assert dict(ctx.seen) == before.seen
        assert np.array_equal(ctx.killers,np.array(before.killers))
        for slot,entry in enumerate(before.tt.table):
            if entry is None:
                assert ctx.flags[slot] == -1,(index,slot)
            else:
                got=(int(ctx.keys[slot]),int(ctx.depths[slot]),int(ctx.scores[slot]),
                     int(ctx.flags[slot]),int(ctx.moves[slot]))
                assert got == entry,(index,slot,entry,got)
        for (color,f,t),value in before.history.items():
            assert ctx.history[int(color),f,t] == value
        assert np.count_nonzero(ctx.history) == len(before.history)
        results.append(record)
    out=WORK/'compiled-search-probe.json'
    out.write_text(json.dumps({'compile_seconds':compile_seconds,'comparisons':results,
        'before_seconds':sum(r['before_seconds'] for r in results),
        'after_seconds':sum(r['after_seconds'] for r in results),
        'not_integrated':True,'playing_strength_validated':False},indent=2))


if __name__ == '__main__':
    main()
