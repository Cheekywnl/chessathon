"""Preserve completed width-256 trial evidence, excluding unselected weights."""
from pathlib import Path
import hashlib
import json
import shutil
import time

work = Path(__file__).resolve().parent
repo = work / 'draw-wide256'
out = repo / 'docs/validation/wide256-20260911'
out.mkdir(parents=True, exist_ok=True)
summary = json.loads((repo/'data/runs/wide-quick-evidence.json').read_text())
assert summary['games'] == 20 and summary['legal_pgn_replay']
assert summary['candidate_failure_or_fallback_markers'] == 0
assert summary['baseline_diagnostics'] == []
for pattern in ('wide-*.log', 'wide-*.json', 'wide-quick-*.jsonl',
                'kernel*.json', 'kernel*.log', 'kernel-column-attempt.py.txt',
                'book-storage-check.*', 'terminal-check.*'):
    for path in (repo/'data/runs').glob(pattern):
        if path.is_file():
            shutil.copyfile(path, out/path.name)
training = work/'draw-fused/data/runs'
for name in ('wide256-pilot','wide256-full','wide256-anneal'):
    dest = out/name
    dest.mkdir(exist_ok=True)
    for path in (training/name).glob('*'):
        if path.suffix in ('.json','.jsonl'):
            shutil.copyfile(path,dest/path.name)
    log = training/f'{name}.log'
    if log.exists():
        shutil.copyfile(log,dest/log.name)
shutil.copyfile(Path(__file__),out/'record_wide_trial.py.txt')
