"""Finish the explicitly authorized, bounded deeper-label training experiment."""
import hashlib,json,os,shutil,subprocess,sys,time
from pathlib import Path
import psutil
work=Path(__file__).parent
data=work/'deeper-teacher';repo=work/'two-hour-refinement'
psutil.Process().cpu_affinity([10])
os.environ['OMP_NUM_THREADS']='1';os.environ['MKL_NUM_THREADS']='1'
os.environ['OPENBLAS_NUM_THREADS']='1';os.environ['NUMBA_NUM_THREADS']='1'
os.environ['NUMBA_OPT']='2'
deadline=time.monotonic()+15*60
manifest=data/'labels-broad-0.manifest.json'
while not manifest.exists():
    if time.monotonic()>deadline:raise SystemExit('Annotation did not finish within the training allocation')
    time.sleep(3)
note=json.loads(manifest.read_text())
for field,name in (('labels_sha256','labels-broad-0.csv'),('details_sha256','labels-broad-0.jsonl')):
    assert note[field]==hashlib.sha256((data/name).read_bytes()).hexdigest()
sys.path.insert(0,str(repo))
from tools.halfkp_data import prepare
prepared=data/'prepared'
result=prepare([data/'labels-broad-0.csv'],prepared,250000,None)
result.update({'source':'Deeper offline Stockfish19 labels of existing self-trained-model source positions',
               'offline_only':True,'annotation_manifest':note,
               'selection_sha256':hashlib.sha256((data/'positions-broad.json').read_bytes()).hexdigest(),
               'no_final_match_positions_used':True})
(prepared/'manifest.json').write_text(json.dumps(result,indent=2))
print(json.dumps({'event':'cache_ready','train':result['train_rows'],
                  'heldout':result['validation_rows']}),flush=True)
parent=work/'next-broadteacher/data/runs/teacher-broad-128/best_checkpoint.pt'
output=data/'training'
commands=[
    (work/'next-broadteacher',['-m','tools.finetune_teacher_mix','--parent',str(parent),
        '--prepared',str(work/'halfkp-trial/data/full-cache'),'--teacher',str(prepared),
        '--out',str(output),'--epochs','120','--lr','0.00002','--ft-lr-scale','0.1',
        '--seed','2026091109'],'training.log'),
    (repo,['-m','tools.quantize_halfkp','--float',str(output/'best_float.npz'),
        '--out',str(data/'quantized-deflate.npz'),'--dense-scaling','fullrange'],'quantize.log'),
    (repo,['-m','tools.repack_halfkp','--source',str(data/'quantized-deflate.npz'),
        '--out',str(data/'halfkp.npz'),'--compression','bzip2'],'repack.log'),
    (repo,['-m','tools.halfkp_quant_check','--float',str(output/'best_float.npz'),
        '--quantized',str(data/'halfkp.npz'),
        '--heldout',str(work/'halfkp-trial/data/full-cache/heldout_fens.csv'),
        '--positions','10000','--cpu','10','--out',str(data/'quant-check.json')],'quant-check.log'),
]
for cwd,args,logname in commands:
    with (data/logname).open('w') as log:
        subprocess.run([sys.executable,'-u',*args],cwd=cwd,stdout=log,stderr=subprocess.STDOUT,check=True)
    print(logname+' completed',flush=True)
target=work/'twohour-deepteacher'
assert not target.exists();target.mkdir()
base=work/'final-extracted-repacked'
for path in base.glob('*.py'):shutil.copy2(path,target/path.name)
for directory in ('weights','syzygy','book'):shutil.copytree(base/directory,target/directory)
shutil.copy2(data/'halfkp.npz',target/'weights/halfkp.npz')
note={'ready_for_matches':True,'runtime':str(target),'model_sha256':hashlib.sha256((data/'halfkp.npz').read_bytes()).hexdigest(),
      'old_model_sha256':'c3207d459bb089de8ba4207c4189d5e04da759f97941d4bd2f6b62d3f3bed768',
      'training_run':str(output/'run.json'),'quantization_check':str(data/'quant-check.json'),
      'playing_strength_validated':False}
(data/'ready.json').write_text(json.dumps(note,indent=2))
print(json.dumps(note,indent=2),flush=True)
