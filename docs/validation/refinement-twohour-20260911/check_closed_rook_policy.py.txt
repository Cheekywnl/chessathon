import hashlib,json,sys,time
from pathlib import Path
import chess,chess.syzygy,psutil
root=Path(__file__).resolve().parent
psutil.Process().cpu_affinity([0])
sys.path.insert(0,str(root/'twohour-rookclosed'))
import chess_rook_endgame as policy
oracle=chess.syzygy.open_tablebase(str(root/'twohour-rookclosed/syzygy'))
oracle.add_directory(str(root/'endgame-oracle'))
started=time.monotonic(); count=0; edges=0; promotions=0
for key,(packed,dtz) in policy._WINS.items():
    board=chess.Board(None)
    for index,(kind,color) in enumerate(((chess.KING,True),(chess.ROOK,True),(chess.PAWN,True),(chess.KING,False),(chess.ROOK,False))):
        board.set_piece_at((key>>(6*index))&63,chess.Piece(kind,color))
    assert board.is_valid() and oracle.probe_wdl(board)==2
    original=board.fen()
    move=policy.winning_move(board)
    assert move is not None and move in board.legal_moves
    zeroing=board.is_zeroing(move)
    board.push(move)
    assert oracle.probe_wdl(board)==-2
    distance=1 if zeroing else abs(oracle.probe_dtz(board))+1
    assert distance==dtz
    if move.promotion:
        assert move.promotion==chess.QUEEN
        promotions+=1
    if board.occupied.bit_count()==5 and board.pawns:
        for reply in board.legal_moves:
            board.push(reply)
            if board.occupied.bit_count()==5 and board.pawns and not board.is_game_over():
                following=policy.winning_move(board)
                assert following is not None,(board.fen(),reply)
                edges+=1
            board.pop()
    board.pop()
    assert board.fen()==original
    variants=[(board,0),(board.transform(chess.flip_horizontal),7),(board.mirror(),56),(board.mirror().transform(chess.flip_horizontal),63)]
    for transformed,mask in variants:
        actual=policy.winning_move(transformed)
        expected=chess.Move(move.from_square^mask,move.to_square^mask,promotion=move.promotion)
        assert actual==expected and actual in transformed.legal_moves
        transformed.halfmove_clock=100-dtz
        assert policy.winning_move(transformed) is None
        transformed.halfmove_clock=99-dtz
        assert policy.winning_move(transformed)==expected
        count+=1
assert policy.winning_move(chess.Board()) is None
assert policy.winning_move(chess.Board('4k3/8/8/8/8/8/8/3RK3 w - - 0 1')) is None
result=dict(passed=True,entries=len(policy._WINS),colour_and_file_symmetry_cases=count,opponent_reply_edges_verified=edges,queen_promotions=promotions,all_wdl_and_dtz_verified=True,all_policy_children_covered=True,fifty_move_boundary_checks=2*count,seconds=time.monotonic()-started)
(root/'twohour-rook-closed-check.json').write_text(json.dumps(result,indent=2))
print(json.dumps(result,indent=2))
