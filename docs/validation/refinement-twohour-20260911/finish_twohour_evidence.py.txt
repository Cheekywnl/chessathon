"""Finalize completed evidence only; does not alter either frozen test runtime."""
import hashlib,json,math,shutil,subprocess,sys
from pathlib import Path
work=Path(__file__).parent;repo=work/'two-hour-refinement'
prefix=repo/'data/runs/twohour-final-vs-current-fullclock'
result=json.loads(prefix.with_suffix('.result.json').read_text())
checks=json.loads((work/'twohour-final-late-checks.json').read_text())
assert result['games']==48 and result['pairs']==24
assert result['score']>0.5,'A nonpositive result requires an explicit review before promotion'
assert checks['endgame_regression_passed'] and checks['wac_passed']>=270
expected='dc545883006320731603675f07c9f75263a60268fb3cc4be7ed9234498204adf'

assert result['configuration']['builds']['opponent']['runtime_sha256']==baseline
for module,suffix in (('tools.match_evidence','.evidence.json'),(str(work/'audit_draw_matches.py'),'.draws.json')):
    if prefix.with_suffix(suffix).exists():
        continue
    cmd=[sys.executable,'-m',module] if module.startswith('tools.') else [sys.executable,module]
    with (work/f'twohour-final{suffix}.log').open('w') as log:
        subprocess.run([*cmd,str(prefix.with_suffix('.jsonl')),'--out',str(prefix.with_suffix(suffix))],cwd=repo,stdout=log,stderr=subprocess.STDOUT,check=True)
evidence=json.loads(prefix.with_suffix('.evidence.json').read_text())
draws=json.loads(prefix.with_suffix('.draws.json').read_text())
assert hashlib.sha256(prefix.with_suffix('.jsonl').read_bytes()).hexdigest() in evidence['log_sha256'].values()
assert draws['games']==result['games']
assert evidence['candidate_failure_or_fallback_markers']==0 and evidence['legal_pgn_replay']
assert evidence['maximum_recorded_init_seconds']<90
assert evidence['maximum_recorded_peak_rss_bytes']<2_000_000_000
rows=[json.loads(line) for line in prefix.with_suffix('.jsonl').read_text().splitlines() if line]
assert not any(row.get('screening_only') for row in rows)
assert all(row['base_ms']==120000 and row['increment_ms']==500 for row in rows)
assert not any(row.get(side+'_init_diagnostics',{}).get('screening_process_reused') for row in rows for side in ('white','black'))
archive=repo/'docs/validation/refinement-twohour-20260911'
for path in prefix.parent.glob(prefix.name+'.*'):shutil.copy2(path,archive/path.name)
for name in ('twohour-final-fullclock.log','twohour-final-wac.log','twohour-final-endgame.log','twohour-final-late-checks.json','twohour-final-late-checks.log','twohour-final.evidence.json.log','twohour-final.draws.json.log'):
    shutil.copy2(work/name,archive/name)
for name in ('twohour-stalemate-oracle.json','twohour-stalemate-review.json','twohour-stalemate-alternative-oracle.json'):
    shutil.copy2(work/name,archive/name)
shutil.copy2(work/'record_twohour_stalemate_review.py',archive/'record_twohour_stalemate_review.py.txt')
shutil.copy2(Path(__file__),archive/(Path(__file__).name+'.txt'))
score=result['score'];elo=400*math.log10(score/(1-score))
summary={'engine_source_commit':'e3a99a808e49374ce130182849448d4724dce588','games':result['games'],'pairs':result['pairs'],'wins':result['wins'],'draws':result['draws'],'losses':result['losses'],'score':score,'descriptive_score_equivalent_elo':elo,'decision':result['decision'],'runtime_sha256':expected,'opponent_runtime_sha256':baseline,'all_fresh_processes':True,'maximum_init_seconds':evidence['maximum_recorded_init_seconds'],'maximum_peak_rss_bytes':evidence['maximum_recorded_peak_rss_bytes'],'final_wac':checks['wac_passed'],'endgame_regression_passed':True,'draw_audit':{k:v for k,v in draws.items() if k!='draw_details'},'platform_elo_established':False,'requested_200_to_500_gain_established':False}
(work/'twohour-final-summary.json').write_text(json.dumps(summary,indent=2)+'\n')
(archive/'final-summary.json').write_text(json.dumps(summary,indent=2)+'\n')
doc=repo/'docs/TWO_HOUR_REFINEMENT_20260911.md';text=doc.read_text()
text=text.replace('The foundation passed 277/300 WAC positions at one second each, the Lucena','The foundation passed 277/300 WAC positions at one second each, the Lucena')
text=text.replace('Its paired comparison against the exact corrected starting engine is in progress.', 'Its completed quick paired comparison against the exact corrected starting engine is recorded below.')
text=text.replace('Its paired comparison against the exact corrected starting engine is in progress.','Its completed quick paired comparison is recorded below.')
text=text.replace('This confirmation has not completed at this checkpoint.','This confirmation is complete; all 24 predeclared pairs are retained below.')
text=text.replace('The full-clock result remains pending at this source checkpoint.','The full-clock result is recorded below.')
