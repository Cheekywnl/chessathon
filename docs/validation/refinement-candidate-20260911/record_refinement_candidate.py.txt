import hashlib

import subprocess

work=Path(__file__).resolve().parent
repo=work/'refinement-hour'
out=repo/'docs/validation/refinement-candidate-20260911'
out.mkdir(parents=True,exist_ok=True)
for name in ('refinement-gate.log','refinement-gate-v2.log','refinement-endgame.log',
             'refinement-draw.json','refinement-draw.log','refinement-reuse.json',

    shutil.copyfile(work/'search-seen-bound/data/runs'/name,out/name)

             'seen-memory-after.log','tt-capacity-probe.json','tt-capacity-probe.log',
             'tablebase-coverage-probe.json','rules-hour.md','contract-hour.md'):
    shutil.copyfile(work/name,out/name)

             'implement_seen_bound.py','assemble_refinement.py','prepare_refinement_screen.py',
             'prepare_incremental_screen.py','record_refinement_candidate.py',
             'implement_pruning_trial.py','split_pruning_trials.py','prepare_pruning_reference.py'):
    shutil.copyfile(work/name,out/(name+'.txt'))

                      ('search-check-depth','check-depth-*')]:
    for p in (work/trial/'data/runs').glob(pattern):
        shutil.copyfile(p,out/p.name)
    subprocess.run(['git','-C',str(work/trial),'diff','--binary',
                    '--output='+str(out/(trial+'.patch')),'--','chess_search.py',
                    'chess_search_compiled.py'],check=True)
shutil.copytree(work/'search-no-ponder/docs/validation/no-ponder-20260911',
                repo/'docs/validation/no-ponder-20260911',dirs_exist_ok=True)
shutil.copyfile(work/'search-no-ponder/docs/NO_PONDER_20260911.md',
                repo/'docs/NO_PONDER_20260911.md')
before=json.loads((work/'seen-memory-before.json').read_text())
after=json.loads((work/'seen-memory-after.json').read_text())
speed=sum(r['seconds'] for r in before)/sum(r['seconds'] for r in after)
assert [(r['move'],r['score'],r['nodes']) for r in before]==[(r['move'],r['score'],r['nodes']) for r in after]
