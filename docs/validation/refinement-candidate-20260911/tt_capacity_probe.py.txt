"""Time bounded iterative searches at two table capacities on old openings."""
import gc
import json
import sys
import time
from pathlib import Path

work = Path(__file__).resolve().parent
repo = work / 'search-accumulator'
sys.path.insert(0, str(repo))
import agent
import chess
import chess_search as cs

positions = json.loads((work / 'halfkp-trial/data/runs/openings-final-20260914.json').read_text())['positions'][:12]
rows = []
for index, row in enumerate(positions):
    board = chess.Board(row['fen'])
    for power in ([21, 22] if index % 2 else [22, 21]):
        table = cs.TranspositionTable(power)
        search = cs.Search(table, {cs.hash_of_board(board): 1})
        started = time.perf_counter()
        deadline = time.monotonic() + 15
        last_score = None
        completed = 0
        for depth in range(1, 10):
            try:
                move, score, _ = search.search_root(board, depth, deadline, prev_score=last_score)
            except cs.TimeUp:
                break
            completed, last_score, uci = depth, score, move.uci()
        result = dict(index=index, power=power, completed_depth=completed,
                      move=uci, score=last_score, nodes=search.nodes,
                      seconds=time.perf_counter()-started)
        rows.append(result)
        print(json.dumps(result), flush=True)
        del search, table
        gc.collect()
out = dict(depth=9, positions=12, rows=rows,
           note='Exploratory timing on previously used openings. Capacity changes can alter search; not a playing-strength test.')
(work/'tt-capacity-probe.json').write_text(json.dumps(out, indent=2)+'\n')
