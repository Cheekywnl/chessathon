from pathlib import Path

work = Path(__file__).resolve().parent
for name, mode in [('search-null-gate', 'null'), ('search-check-depth', 'check')]:
    for filename in ('chess_search.py', 'chess_search_compiled.py'):
        source = (work/'search-pruning'/filename).read_text()
        if mode == 'null':
            source = source.replace('            and (not childcheck)\n        ):\n            reduction = min(',
                                    '        ):\n            reduction = min(')
            source = source.replace('                and not child_in_check\n            ):\n                d_idx =',
                                    '            ):\n                d_idx =')
        else:
            if filename == 'chess_search_compiled.py':
                source = source.replace('''    # A null move is a cutoff probe: only try it when the static score already
    # reaches beta. Preserve real search effort in positions that need improving.
    if (allow_null and not incheck and depth >= 3 and static >= beta
            and own & ~s[0] & ~s[5]):''',
                    '    if allow_null and (not incheck) and (depth >= 3) and own & ~s[0] & ~s[5]:')
            else:
                source = source.replace('''        if (allow_null and not in_check and depth >= 3
                and static_eval is not None and static_eval >= beta
                and has_non_pawn_material(state)):''',
                    '        if allow_null and not in_check and depth >= 3 and has_non_pawn_material(state):')
        (work/name/filename).write_text(source)
    (work/name/'data/runs').mkdir(parents=True, exist_ok=True)
