"""Audit all iterative states, then summarize the complete ABBA timings."""
import json
from pathlib import Path

root = Path(__file__).resolve().parent
names = ["before-a", "after-a", "after-b", "before-b"]
runs = {name: json.loads((root / f"scratch-{name}.json").read_text()) for name in names}
reference = runs["before-a"]
for name, run in runs.items():
    assert run["nodes"] == reference["nodes"]
    for actual, expected in zip(run["records"], reference["records"], strict=True):
        for key in ("fen", "depths", "nodes", "table_and_heuristics_sha256"):
            assert actual[key] == expected[key], (name, actual["index"], key)
before = sum(runs[name]["search_seconds"] for name in ("before-a", "before-b"))
after = sum(runs[name]["search_seconds"] for name in ("after-a", "after-b"))
result = {"order": names, "positions": len(reference["records"]), "depth": reference["depth"],
          "nodes_each_run": reference["nodes"], "mismatches": 0,
          "before_seconds": before, "after_seconds": after, "speed_ratio": before / after,
          "seconds_by_run": {n: r["search_seconds"] for n, r in runs.items()},
          "playing_strength_validated": False}
(root / "scratch-speed-summary.json").write_text(json.dumps(result, indent=2))
print(json.dumps(result, indent=2))
