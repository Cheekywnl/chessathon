"""Merge immutable teacher evidence, filtering and deduplicating before preparation."""
import csv
import hashlib
import json
import shutil
import subprocess
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent
SOURCE = ROOT / "teacher-tools"
sys.path.insert(0, str(SOURCE))
from annotate_broad_positions import promotion_budget_valid
import chess

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

destination = ROOT / "next-broadteacher"
if not destination.exists():
    subprocess.run(["git", "worktree", "add", "-b", "refine/broadteacher-next",
                    str(destination), "ce3bdbb260fa714fecf4279f5befe6bce3ba1c4e"],
                   cwd=ROOT / "hour-release", check=True)
assert subprocess.check_output(["git", "branch", "--show-current"], cwd=destination,
                               text=True).strip() == "refine/broadteacher-next"
for path in (ROOT / "next-scratch").glob("*.py"):
    shutil.copy2(path, destination / path.name)
for name in ("finetune_teacher_mix.py", "train_phaseheads.py", "platform_agent.py",
             "sprt_arena.py"):
    shutil.copy2(ROOT / "next-teacher/tools" / name, destination / "tools" / name)
(destination / "data/runs").mkdir(parents=True, exist_ok=True)
merged = {}
notes = []
rejected = []
duplicates = 0
for stem in ("labels-0", "labels-1", "labels-broad-0", "labels-broad-1"):
    note = json.loads((SOURCE / f"{stem}.manifest.json").read_text())
    assert note["labels_sha256"] == sha(SOURCE / f"{stem}.csv")
    assert note["details_sha256"] == sha(SOURCE / f"{stem}.jsonl")
    input_name = "positions-broad.json" if "broad" in stem else "positions.json"
    assert note["input_sha256"] == sha(SOURCE / input_name)
    notes.append(note)
    with (SOURCE / f"{stem}.csv").open(newline="", encoding="utf8") as stream:
        for row in csv.DictReader(stream):
            board = chess.Board(row["fen"])
            if not board.is_valid() or not promotion_budget_valid(board):
                rejected.append({"source": stem, **row})
                continue
            key = " ".join(row["fen"].split()[:4])
            if key in merged:
                duplicates += 1
                if int(row["depth"]) <= int(merged[key]["depth"]):
                    continue
            merged[key] = row
out = SOURCE / "labels-merged.csv"
with out.open("w", newline="", encoding="utf8") as stream:
    writer = csv.DictWriter(stream, fieldnames=["fen", "depth", "cp"])
    writer.writeheader()
    writer.writerows(merged.values())
(SOURCE / "labels-merged.rejected.json").write_text(json.dumps(rejected, indent=2))
sys.path.insert(0, str(destination))
from tools.halfkp_data import prepare
cache = destination / "data/teacher-cache"
manifest = prepare([out], cache, 250000, None)
manifest.update({"source": "Own game and sampled Lichess positions annotated offline by Stockfish 19",
                 "teacher_annotations": notes, "offline_only": True,
                 "merge_source_sha256": sha(Path(__file__)),
                 "rules_snapshot_sha256": sha(SOURCE / "rules-20260911.md"),
                 "duplicates_removed": duplicates, "promotion_budget_rejected": len(rejected),
                 "selection": "Quiet best move, depth >=6, 12000 nodes, abs(cp)<=1800; hash heldout split unchanged"})
(cache / "manifest.json").write_text(json.dumps(manifest, indent=2))
print(json.dumps({"train": manifest["train_rows"], "heldout": manifest["validation_rows"],
                  "duplicates": duplicates, "rejected": len(rejected),
                  "manifest_sha256": sha(cache / "manifest.json")}, indent=2))
